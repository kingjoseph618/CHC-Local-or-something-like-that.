# Copy and run in PowerShell
$Root = "$env:USERPROFILE\Desktop\JAMIE_AI_INTAKE_001"
$Zip  = "$env:USERPROFILE\Desktop\JAMIE_AI_INTAKE_001.zip"

New-Item -ItemType Directory -Force -Path $Root | Out-Null

Get-ChildItem -Path $Root -Recurse -File -ErrorAction SilentlyContinue |
    Select-Object FullName, Name, Extension, Length, CreationTime, LastWriteTime |
    Export-Csv "$Root\FILE_MANIFEST.csv" -NoTypeInformation -Encoding UTF8

@"
JAMIE AI INTAKE
Generated: $(Get-Date)

Purpose:
Package files for inspection.

Rules:
- No files deleted
- No files moved
- No files executed
- No passwords/API keys/tokens included
"@ | Out-File "$Root\README.txt" -Encoding UTF8

if (Test-Path $Zip) {
    Remove-Item $Zip -Force
}

Compress-Archive -Path "$Root\*" -DestinationPath $Zip

Write-Host "INTAKE PACKAGE CREATED SUCCESSFULLY AT: $Zip" -ForegroundColor Cyan
